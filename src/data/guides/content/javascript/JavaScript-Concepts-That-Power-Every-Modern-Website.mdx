
## 1. **The DOM (Document Object Model)**
The interface between JavaScript and HTML - every interaction with the page goes through the DOM.

```javascript
// Querying
const element = document.querySelector('.button');
const elements = document.querySelectorAll('.item');

// Manipulation
element.textContent = 'New text';
element.classList.add('active');
element.setAttribute('data-id', '123');

// Creation
const div = document.createElement('div');
div.innerHTML = '<p>Hello</p>';
document.body.appendChild(div);

// 2026: DOM with declarative shadow DOM
const host = document.querySelector('#app');
const shadow = host.attachShadow({ mode: 'open' });
shadow.innerHTML = `
  <style>h1 { color: red; }</style>
  <h1>Scoped styles</h1>
`;
```

## 2. **Event-Driven Architecture**
Everything in the browser is event-driven - clicks, scrolls, network responses, timers.

```javascript
// Event listeners
element.addEventListener('click', handleClick);

// Event delegation (bubbling)
document.addEventListener('click', (e) => {
  if (e.target.matches('.button')) {
    handleButtonClick(e);
  }
});

// Custom events
const event = new CustomEvent('userLogin', { 
  detail: { userId: 123 } 
});
document.dispatchEvent(event);

// 2026: AbortController for cleanup
const controller = new AbortController();
element.addEventListener('click', handler, { signal: controller.signal });
// controller.abort() removes ALL listeners
```

## 3. **Asynchronous Programming (Promises/Async)**
Every modern website depends on async operations - loading data, rendering, user interactions.

```javascript
// The foundation of modern web
async function loadData() {
  try {
    const response = await fetch('/api/data');
    const data = await response.json();
    renderData(data);
  } catch (error) {
    handleError(error);
  }
}

// Concurrent loading
const [user, posts] = await Promise.all([
  fetchUser(),
  fetchPosts()
]);

// Streaming responses (2026)
const response = await fetch('/api/large-data');
const reader = response.body.getReader();
while (true) {
  const { done, value } = await reader.read();
  if (done) break;
  processChunk(value);
}
```

## 4. **Reactive State Management**
Every modern framework (React, Vue, Svelte) is built on reactivity.

```javascript
// Simple reactivity with Proxies
function reactive(obj) {
  return new Proxy(obj, {
    set(target, key, value) {
      target[key] = value;
      // Trigger re-render
      updateDOM();
      return true;
    }
  });
}

// Signals (the new trend in 2026)
const count = signal(0);
const double = computed(() => count() * 2);

effect(() => {
  console.log(`Count: ${count()}, Double: ${double()}`);
});

count.set(5); // Automatically logs new values
```

## 5. **Component-Based Architecture**
Sites are built from reusable components, not monolithic pages.

```javascript
// Web Components (native)
class UserCard extends HTMLElement {
  static observedAttributes = ['user-id'];
  
  async connectedCallback() {
    const user = await fetchUser(this.getAttribute('user-id'));
    this.innerHTML = `
      <div class="card">
        <h2>${user.name}</h2>
        <p>${user.email}</p>
      </div>
    `;
  }
}

customElements.define('user-card', UserCard);

// React component with hooks
function UserCard({ userId }) {
  const [user, setUser] = useState(null);
  
  useEffect(() => {
    fetchUser(userId).then(setUser);
  }, [userId]);
  
  if (!user) return <Loading />;
  return <div>{user.name}</div>;
}
```

## 7. **Performance Optimization (Code Splitting, Lazy Loading)**
Modern sites load only what's needed, when it's needed.

```javascript
// Route-based code splitting
const routes = {
  '/dashboard': () => import('./Dashboard.js'),
  '/profile': () => import('./Profile.js'),
  '/admin': () => import('./Admin.js')
};

// Intersection Observer for lazy loading images
const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src;
      observer.unobserve(img);
    }
  });
});

document.querySelectorAll('img[data-src]').forEach(img => {
  observer.observe(img);
});

// Priority hints (2026)
<link rel="preload" as="image" href="hero.jpg" fetchpriority="high">
```

## 8. **Service Workers & PWA**
Sites work offline, load instantly, and can send push notifications.

```javascript
// Service worker registration
if ('serviceWorker' in navigator) {
  navigator.serviceWorker.register('/sw.js');
}

// sw.js - Cache API
const CACHE_NAME = 'v1';
const urlsToCache = ['/', '/styles.css', '/app.js'];

self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(urlsToCache))
  );
});

self.addEventListener('fetch', (event) => {
  event.respondWith(
    caches.match(event.request)
      .then(response => response || fetch(event.request))
  );
});

// Background sync (2026)
const sync = await navigator.serviceWorker.ready;
await sync.sync.register('sync-data');
```

## 10. **Security Features (CSP, Trusted Types)**
Modern sites implement multiple layers of security.

```javascript
// Content Security Policy
// meta tag or server header
<meta http-equiv="Content-Security-Policy" 
      content="default-src 'self'; script-src 'self' trusted-cdn.com">

// Trusted Types (prevents XSS)
if (window.trustedTypes) {
  const policy = trustedTypes.createPolicy('my-policy', {
    createHTML: (input) => {
      // Sanitize input
      return sanitize(input);
    }
  });
  
  element.innerHTML = policy.createHTML('<div>Safe</div>');
}

// SameSite cookies
document.cookie = 'session=123; SameSite=Strict; Secure';

// CORS
fetch('/api', {
  headers: {
    'Origin': window.location.origin
  }
});
```
